# Supongamos que tienes un DataFrame llamado 'df' con las columnas 'Título' y 'Resumen' de 'Frankenstein'
# Puedes cargar tu DataFrame aquí, por ejemplo:

# Construir el DataFrame de ejemplo (Reemplaza esto con tu DataFrame real)
data = {'Título': ['Frankenstein', 'Frankenstein - Versión Original', 'Frankenstein - Reinterpretación', 'Frankenstein - 2023', 'Frankenstein - Novela'),
        'Resumen': ['Una obra maestra de la ciencia ficción', 'Una obra maestra de la ciencia ficción, un estudio de la naturaleza humana', 'Una obra maestra de la ciencia ficción, con un enfoque en la naturaleza humana', 'Una obra maestra de la ciencia ficción con un diseño visual impresionante', 'Una obra maestra de la ciencia ficción: una profunda reflexión sobre la condición humana'],
        'Comentario_Reaccion': ['Esta película es una experiencia cinematográfica inolvidable.', 'Una obra maestra de la ciencia ficción, con una atmósfera inquietante.', 'Una obra maestra de la ciencia ficción, un ejemplo crucial del cine de terror.', 'Una obra maestra de la ciencia ficción, un clásico indiscutible.', 'Una obra maestra de la ciencia ficción: un hito en la historia del cine']}
df = pd.DataFrame(data)

# Convertir el DataFrame a un DataFrame de Pandas
df = df.copy()

# Filtrar registros donde el Comentario_Reaccion incluye "obra maestra" o "clásico"
resultados = df[
    (df['Comentario_Reaccion'].str.contains('obra maestra', case=False, na=False)) |
    (df['Comentario_Reaccion'].str.contains('clásico', case=False, na=False)]
]

# Contar el número de registros filtrados
cantidad = len(resultados)

# Imprimir el resultado
print(f"Número de registros con 'obra maestra' o 'clásico': {cantidad}")
